# helios-payables runtime dependencies. All pinned.

# The four platform packages are NOT listed: they are imported in place from
# ../../platform/*/src, so this app runs with no install step and cannot drift from
# the versions its teammates are editing.

# `jsonschema` is the one that matters most and the easiest to miss. `audit_agent`
# imports it lazily inside a try/except (policy.py:416) and, when it is absent,
# DENIES EVERY SIDE EFFECT with "argument schema validation is unavailable". That is
# the correct fail-closed behaviour, but it means a missing dependency presents as
# "the policy is strict" rather than "the app is broken" — every payment refused, no
# traceback. It is a declared dependency of both action-agent-builder and
# agent-integration, so it is not a new third-party addition.

pydantic==2.12.4
PyYAML==6.0.3
